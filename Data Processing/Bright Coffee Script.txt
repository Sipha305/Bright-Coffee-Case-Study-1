SELECT
    transaction_id,
     CAST(REPLACE(unit_price,',','.')AS DOUBLE) AS unit_price
FROM retail.sales.bright_coffee_shop_sales;    

-- Checking the number of Records
SELECT
     COUNT(*) AS total_records
FROM retail.sales.bright_coffee_shop_sales;

-- Checking for duplicates 
SELECT*,
      COUNT(*) AS duplicate_count
FROM retail.sales.bright_coffee_shop_sales
GROUP BY ALL
HAVING COUNT(*) >1;

-- Checking for NULL values
SELECT*
FROM retail.sales.bright_coffee_shop_sales
WHERE transaction_id IS NULL OR transaction_date IS NULL OR transaction_time IS NULL OR
transaction_qty IS NULL OR store_id IS NULL OR store_location IS NULL OR product_id IS NULL OR unit_price
IS NULL OR product_id IS NULL OR product_id IS NULL OR product_detail IS NULL;

-- 1. Total Revenue
SELECT
      SUM(transaction_qty * CAST(REPLACE(unit_price,',','.')AS DECIMAL(10,2)) ) AS total_revenue 
FROM retail.sales.bright_coffee_shop_sales;

-- 7. TOTAL REVEUNE BASED IN THE TIME OF THE DAY
    --- Earliest transaction(Time Buckets) 
    SELECT 
         MIN(transaction_time),
         MAX(transaction_time)
    FROM retail.sales.bright_coffee_shop_sales;

    --06am - 11:59:59 > Morning
    -- 12pm - 16:59:59 > Afternoon
    -- 5pm - 19:59:59 > Evening
    -- After 20:00 > Night

SELECT 
     CASE
         WHEN transaction_time BETWEEN '06:00:00' AND '11:59:59' THEN 'Morning'
         WHEN transaction_time BETWEEN '12:00:00' AND '16:59:59' THEN 'Afternoon'
         WHEN transaction_time BETWEEN '17:00:00' AND '19:59:59' THEN 'Evening'
         ELSE 'Night'
         END AS time_bucket
FROM retail.sales.bright_coffee_shop_sales;
